## 4. Explicación técnica

Sanitización determinista de variables protegidas: Antes de enviar el perfil a Jev, un script en CPU elimina nombres, fotografías, fechas de nacimiento, direcciones residenciales y nombres de instituciones educativas que actúan como variables proxy socioeconómicas.

import re, logging

logger = logging.getLogger("P16_05")

def anonimizar_perfil_laboral(cv_texto_crudo: str) -> str:
    try:
        # Eliminar edad, fechas de nacimiento y genero
        texto_limpio = re.sub(r"\b\d{1,2}\s*(?:años|anos)\b", "[EDAD_OMITIDA]", cv_texto_crudo, flags=re.IGNORECASE)
        texto_limpio = re.sub(r"\b(?:femenino|masculino|soltero|casado|divorciado)\b", "[ESTADO_OMITIDO]", texto_limpio, flags=re.IGNORECASE)
        texto_limpio = re.sub(r"\b(?:comuna|domicilio|residencia)\s*:\s*[^\n]+", "[DIRECCION_OMITIDA]", texto_limpio, flags=re.IGNORECASE)
        return texto_limpio
    except Exception as err:
        logger.error(f"Fallo en anonimizacion laboral: {type(err).__name__} (detalles omitidos por seguridad)")
        return ""
