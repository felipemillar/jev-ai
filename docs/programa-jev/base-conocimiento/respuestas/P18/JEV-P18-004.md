def calcular_prioridad_oportunidad(valor: float, factibilidad: float, evidencia: float, riesgo: float, costo: float) -> float:
    try:
        puntaje = (valor * 0.30) + (factibilidad * 0.25) + (evidencia * 0.20) - (riesgo * 0.15) - (costo * 0.10)
