## 5. Ejemplo sintético trabajado y contraejemplo o caso límite

### 5.1 Implementación en Python (SDK 0.7.0)

```python
import logging
from typesafe_sdk import TypeSafeClient, Choice, Score
logger = logging.getLogger('P08_19')

def flujo_enrutamiento_integral(texto_solicitud: str) -> dict:
    try:
        client = TypeSafeClient(model='jev-1.13.0')
        res = client.system_one(state=texto_solicitud, questions={'area': Choice(instructions='Departamento', criteria={'it': 'Departamento de Tecnologías de la Información', 'legal': 'Departamento Legal y Regulatorio', 'rrhh': 'Departamento de Recursos Humanos'}), 'prioridad': Score(instructions='Severidad 1-3', criteria=['Nivel 1: Nivel inicial con fiabilidad baja', 'Nivel 2: Nivel regular con reservas', 'Nivel 3: Nivel aceptable conforme', 'Nivel 4: Nivel óptimo con fiabilidad alta'])})
        return {'departamento': res.answers['area'].choice, 'score_severidad': res.answers['prioridad'].score, 'escalar_inmediato': res.answers['prioridad'].score >= 2.6}
    except Exception as err:
        logger.error(f'Fallo enrutamiento: {type(err).__name__} (detalles omitidos por seguridad)')
        return {'error': 'Fallo_Flujo'}
```
