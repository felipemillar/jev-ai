## 5. Ejemplo sintético trabajado y contraejemplo o caso límite

### 5.1 Implementación en Python (SDK 0.7.0)

```python
# Ejemplo ilustrativo no ejecutado
import logging
from typesafe_sdk import TypeSafeClient, Choice, Score

logger = logging.getLogger("P08_19")

def flujo_enrutamiento_integral(texto_solicitud: str) -> dict:
    try:
        client = TypeSafeClient(model="jev-1.13.0")
        res = client.system_one(
            state=texto_solicitud,
            questions={
                "area": Choice(instructions="Departamento", options=["it", "legal", "rrhh"]),
                "prioridad": Score(instructions="Severidad 1-3", anchors=[{"level": 1, "label": "baja", "description": "leve"}, {"level": 3, "label": "alta", "description": "critica"}])
            }
        )
        return {
            "departamento": res.answers["area"].choice,
            "score_severidad": res.answers["prioridad"].score,
            "escalar_inmediato": res.answers["prioridad"].score >= 2.6
        }
    except Exception as err:
        logger.error(f"Fallo enrutamiento: {type(err).__name__} (detalles omitidos por seguridad)")
        return {"error": "Fallo_Flujo"}
```

Construir un flujo de enrutamiento que dependa de Jev sin prever una cola de desborde para peticiones no clasificadas provocará la pérdida silenciosa de tickets.
