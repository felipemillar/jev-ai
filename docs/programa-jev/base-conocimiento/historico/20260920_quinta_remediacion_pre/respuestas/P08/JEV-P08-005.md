### 5.1 Implementación en Python (SDK 0.7.0)

```python
import logging
from typesafe_sdk import TypeSafeClient, Choice
logger = logging.getLogger('P08_05')

def validar_aserciones_contrato(texto_cv: str, anios_declarados: int) -> str:
    try:
        client = TypeSafeClient(model='jev-1.13.0')
        res = client.system_one(state=texto_cv, questions={'rango': Choice(instructions='Seniority', criteria={'junior': 'Perfil profesional junior inicial', 'senior': 'Perfil profesional senior experimentado'})})
        s = res.answers['rango'].choice
        if s == 'senior' and anios_declarados < 2:
            return 'junior_rectificado_por_invariante'
        return s
    except Exception as err:
        logger.error(f'Fallo en asercion: {type(err).__name__} (detalles omitidos por seguridad)')
        return 'indeterminado'
```
